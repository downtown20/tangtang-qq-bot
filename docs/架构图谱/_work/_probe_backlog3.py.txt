# -*- coding: utf-8 -*-
import sqlite3, json
from datetime import datetime
c = sqlite3.connect(r'D:\qq-小糖糖\memory.db')
rows = c.execute("""
SELECT c.qq_id, COUNT(*), MIN(NULLIF(c.timestamp,''))
FROM chat_log c
LEFT JOIN extraction_cursors e ON e.pipeline='memory' AND e.direction='forward' AND e.qq_id=c.qq_id
WHERE COALESCE(c.is_bot_reply,0)=0 AND COALESCE(c.quarantined_at,'')=''
  AND c.id > COALESCE(e.cursor_chat_id,0)
GROUP BY c.qq_id
""").fetchall()
now = datetime.now()
r10=r7=r21=0
near = []
for qq,n,o in rows:
    try: a=(now-datetime.fromisoformat(o)).total_seconds()/86400.0
    except Exception: a=-1
    if n>=10: r10+=1
    elif n>=3 and a>=7: r7+=1
    elif n>0 and a>=21: r21+=1
    else:
        if n>=3: near.append({'qq':qq,'n':n,'age_days':round(a,1)})
print(json.dumps({
 'ready_count_ge10': r10, 'ready_age7_n3': r7, 'ready_age21': r21,
 'near_miss_n_ge3': sorted(near, key=lambda x:-x['n'])[:10],
 'near_miss_total': len(near),
}, ensure_ascii=False, indent=1))
