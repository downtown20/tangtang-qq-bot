# -*- coding: utf-8 -*-
"""只读取证 3：体验缓冲累积速率 vs 反思门槛（autonomy-tracer / task-2）"""
import io, os, json, sys, collections
sys.stdout.reconfigure(encoding='utf-8')
root = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..', '..', '..'))
d = json.load(io.open(os.path.join(root, '状态', '.tangtang_self.json'), encoding='utf-8'))
buf = d.get('experience_buffer') or []
c = collections.Counter((e.get('time') or '')[:10] for e in buf)
print('per-day entries:', sorted(c.items()))
real = [e for e in buf if e.get('type') != 'internal_digest']
cr = collections.Counter((e.get('time') or '')[:10] for e in real)
print('real (non-digest) per-day:', sorted(cr.items()))
upto = lambda day: sum(1 for e in buf if (e.get('time') or '')[:10] <= day)
for day in ['2026-10-12', '2026-10-13', '2026-10-14', '2026-10-15']:
    print('cumulative through', day, '=', upto(day))
