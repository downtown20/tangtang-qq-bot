# -*- coding: utf-8 -*-
"""只读日志取证 2：样本行 + 关键事件统计，结果写 UTF-8 文件避免控制台编码。"""
import io, collections, json
p = r'D:\qq-小糖糖\日志\tangtang.log'
out = []
worker_tail = []
brain_lines = collections.Counter()
sample_brain = []
n_brain = 0
with io.open(p, 'r', encoding='utf-8', errors='replace') as f:
    for line in f:
        if '提取 worker 周期' in line:
            worker_tail.append(line.rstrip()[-220:])
            if len(worker_tail) > 3: worker_tail.pop(0)
        if '\U0001f9e0' in line:
            n_brain += 1
            key = line.split('|')[0][-60:]
            brain_lines[key] += 1
            if len(sample_brain) < 12: sample_brain.append(line.rstrip()[:200])
out.append('brain_marker_lines=' + str(n_brain))
out.append('--- brain line kinds (top 15) ---')
for k, v in brain_lines.most_common(15):
    out.append(f'{v}  {k}')
out.append('--- sample brain lines ---')
out.extend(sample_brain)
out.append('--- last worker cycles ---')
out.extend(worker_tail)
io.open(r'D:\qq-小糖糖\docs\架构图谱\_work\_log_probe2.txt', 'w', encoding='utf-8').write('\n'.join(out))
print('done', n_brain)
