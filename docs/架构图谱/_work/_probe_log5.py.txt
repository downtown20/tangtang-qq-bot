# -*- coding: utf-8 -*-
import io
p = r'D:\qq-小糖糖\日志\tangtang.log'
keys = ['关系档案', '画像补全触发', '画像', '记忆']
out = {k: [] for k in keys}
with io.open(p,'r',encoding='utf-8',errors='replace') as f:
    for line in f:
        for k in keys:
            if k in line and len(out[k]) < 14:
                out[k].append(line.rstrip()[:230])
w = []
for k in keys:
    w.append('=== ' + k + ' ===')
    w.extend(out[k])
io.open(r'D:\qq-小糖糖\docs\架构图谱\_work\_log_probe5.txt','w',encoding='utf-8').write('\n'.join(w))
print('ok')
