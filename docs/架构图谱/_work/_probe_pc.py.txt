# -*- coding: utf-8 -*-
import io, os, sys, datetime
sys.stdout.reconfigure(encoding='utf-8')
root = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), '..', '..', '..'))
for dirpath, dirnames, filenames in os.walk(os.path.join(root, '.pytest_cache')):
    for fn in filenames:
        fp = os.path.join(dirpath, fn)
        print(os.path.relpath(fp, root), datetime.datetime.fromtimestamp(os.path.getmtime(fp)), os.path.getsize(fp))
